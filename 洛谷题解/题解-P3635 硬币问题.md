```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;cin>>n;
    int v[] = {0,1,5,11}; // 四种面额的硬币

    vector<int> f(n+1,0x3f3f3f3f); // 初始化递推数组

    f[0]=0;
    for(int i=1;i<=3;i++){
        for(int j=v[i]; j<=n; j++) {
            f[j] = min(f[j], f[j-v[i]]+1); // 关键
        }
    }
    if(f[n]!=0x3f3f3f3f) cout << f[n];
    else cout<< -1;
}
```

$$f_j = min (f_j,  f_{j-v_i}+1)$$
这一行是最重要的递推式，详见[[推论-完全背包-最小计数问题]]

$v_i$ 既做重量也做价值，求最小
